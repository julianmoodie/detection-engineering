    LogName = "Microsoft-Windows-PowerShell/Operational"
} | Where-Object {
    $_.Message -like "*-enc*" -or

    $_.Message -like "* -iwr*" -or
    $_.Message -like "*Invoke*" -or
    $_.Message -like "*invoke*" -or
    $_.Message -like "*curl*" -or
}
